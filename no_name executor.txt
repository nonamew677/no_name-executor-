local function LoadConfigData()
    pcall(function()
        if type(readfile) == "function" then
            local content = readfile(STORAGE_FILE)
            if content and #content > 0 then
                local decoded = HttpService:JSONDecode(content)
                if decoded and type(decoded) == "table" then
                    if decoded.SavedScripts then
                        for name, data in pairs(decoded.SavedScripts) do
                            if type(data) == "table" and data.Code then
                                SystemConfig.SavedScripts[name] = { Code = data.Code }
                            end
                        end
                    end
                    if decoded.LuaAutoLoadScripts then
                        for name, enabled in pairs(decoded.LuaAutoLoadScripts) do
                            if type(enabled) == "boolean" then
                                SystemConfig.LuaAutoLoadScripts[name] = enabled
                            end
                        end
                    end
                    if type(decoded.CustomName) == "string" and #decoded.CustomName > 0 then
                        SystemConfig.CustomName = decoded.CustomName
                    end
                    if type(decoded.ShineEnabled) == "boolean" then
                        SystemConfig.ShineEnabled = decoded.ShineEnabled
                    end
                    if type(decoded.CrosshairText) == "string" then
                        SystemConfig.CrosshairText = decoded.CrosshairText
                    end
                    if type(decoded.CrosshairEnabled) == "boolean" then
                        SystemConfig.CrosshairEnabled = decoded.CrosshairEnabled
                    end
                    if type(decoded.CrosshairTextEnabled) == "boolean" then
                        SystemConfig.CrosshairTextEnabled = decoded.CrosshairTextEnabled
                    end
                    if type(decoded.RainbowEnabled) == "boolean" then
                        SystemConfig.RainbowEnabled = decoded.RainbowEnabled
                    end
                    if type(decoded.CrosshairStickEnabled) == "boolean" then
                        SystemConfig.CrosshairStickEnabled = decoded.CrosshairStickEnabled
                    end
                    if type(decoded.StickSmoothing) == "number" then
                        SystemConfig.StickSmoothing = decoded.StickSmoothing
                    end
                    if type(decoded.StickFOV) == "number" then
                        SystemConfig.StickFOV = decoded.StickFOV
                    end
                    if type(decoded.StickTeamCheck) == "boolean" then
                        SystemConfig.StickTeamCheck = decoded.StickTeamCheck
                    end
                    if type(decoded.SavedUIPosition) == "table" and decoded.SavedUIPosition.XScale then
                        SystemConfig.SavedUIPosition = decoded.SavedUIPosition
                    end
                    if type(decoded.CrosshairSpinSpeed) == "number" then
                        SystemConfig.CrosshairSpinSpeed = math.clamp(decoded.CrosshairSpinSpeed, -2000, 2000)
                    end
                    if type(decoded.CrosshairPulseSpeed) == "number" then
                        SystemConfig.CrosshairPulseSpeed = math.clamp(decoded.CrosshairPulseSpeed, 0, 50)
                    end
                    if type(decoded.CrosshairOutlineEnabled) == "boolean" then
                        SystemConfig.CrosshairOutlineEnabled = decoded.CrosshairOutlineEnabled
                    end
                    if type(decoded.CrosshairTextOutlineEnabled) == "boolean" then
                        SystemConfig.CrosshairTextOutlineEnabled = decoded.CrosshairTextOutlineEnabled
                    end
                    if decoded.CrosshairMode == "Default" or decoded.CrosshairMode == "Yokai" then
                        SystemConfig.CrosshairMode = decoded.CrosshairMode
                    end
                end
            end
        end
    end)
end

local function SaveConfigData()
    pcall(function()
        if type(writefile) == "function" then
            writefile(STORAGE_FILE, HttpService:JSONEncode(SystemConfig))
        end
    end)
end

-- Core execution logic
local function PureCleanupScript(name: string)
    if ActiveUIElements[name] then
        for _, instance in ipairs(ActiveUIElements[name]) do
            pcall(function() instance:Destroy() end)
        end
        ActiveUIElements[name] = nil
    end
end

local function FastTrackExecuteScript(name: string, code: string)
    PureCleanupScript(name)
    ActiveUIElements[name] = {}

    local initialGuis = playerGui:GetChildren()

    local func, err = loadstring(code)
    if func then
        task.spawn(function()
            local success, runErr = pcall(func)
            if not success then
                warn("[NO_NAME RUNTIME EXCEPTION] " .. tostring(runErr))
            else
                task.wait(0.3)
                for _, child in ipairs(playerGui:GetChildren()) do
                    if child:IsA("ScreenGui") and not table.find(initialGuis, child) and child.Name ~= "no_nameUI" then
                        table.insert(ActiveUIElements[name], child)
                    end
                end
            end
        end)
    else
        warn("[NO_NAME COMPILE FAILURE]: " .. tostring(err))
    end
end

local function RunAutoLoadScripts()
    if not SystemConfig.LuaAutoLoadScripts then return end
    for name, enabled in pairs(SystemConfig.LuaAutoLoadScripts) do
        if enabled == true and SystemConfig.SavedScripts[name] then
            FastTrackExecuteScript(name, SystemConfig.SavedScripts[name].Code)
        end
    end
end

-- ================= CROSSHAIR STICK ENGINE =================
local function WorldToScreen(worldPos: Vector3): Vector2?
    local cam = workspace.CurrentCamera
    if not cam then return nil end
    local rel = cam.CFrame:PointToObjectSpace(worldPos)
    if rel.Z >= 0 then return nil end
    local vp = cam.ViewportSize
    local tanHalf = math.tan(math.rad(cam.FieldOfView) * 0.5)
    local aspect = vp.X / vp.Y
    local sx = (rel.X / (-rel.Z * tanHalf * aspect)) * (vp.X * 0.5)
    local sy = (rel.Y / (-rel.Z * tanHalf)) * (vp.Y * 0.5)
    return Vector2.new(sx, sy)
end

local function getHeadPos(character: Model): Vector3?
    local head = character:FindFirstChild("Head")
    if head and head:IsA("BasePart") then return head.Position end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if hrp and hrp:IsA("BasePart") then return hrp.Position end
    return nil
end

local function isAlive(model: Model?): boolean
    if not model then return false end
    local hum = model:FindFirstChildOfClass("Humanoid")
    return hum ~= nil and hum.Health > 0
end

local function findStickTarget(): (Vector2?, Model?)
    local myTeam = player.Team
    local bestDist, bestScreen, bestModel = SystemConfig.StickFOV, nil, nil
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= player then
            local char = other.Character
            if isAlive(char) then
                local skip = false
                if SystemConfig.StickTeamCheck and myTeam and other.Team == myTeam then
                    skip = true
                end
                if not skip then
                    local head = getHeadPos(char)
                    if head then
                        local screen = WorldToScreen(head)
                        if screen then
                            local d = screen.Magnitude
                            if d < bestDist then
                                bestDist = d
                                bestScreen = screen
                                bestModel = char
                            end
                        end
                    end
                end
            end
        end
    end
    return bestScreen, bestModel
end

local stickOffset = Vector2.new(0, 0)

    local target = screen or Vector2.new(0, 0)
    stickOffset = stickOffset:Lerp(target, math.clamp(dt * SystemConfig.StickSmoothing, 0, 1))
end)

local hue, rotation, pulse = 0, 0, 0

RunService.RenderStepped:Connect(function(dt)
    crosshair.Visible = SystemConfig.CrosshairEnabled
    crosshairLabel.Visible = SystemConfig.CrosshairEnabled and SystemConfig.CrosshairTextEnabled

    if crosshairLabel.Text ~= SystemConfig.CrosshairText then
        crosshairLabel.Text = SystemConfig.CrosshairText
    end

    if not SystemConfig.CrosshairEnabled then
        return
    end

    local effSpin, effPulse, effOutline, tint, tintEdge = GetEffectiveCrosshairParams()

    if SystemConfig.RainbowEnabled and not tint then
        hue = (hue + dt * 0.25) % 1
    end
    rotation = (rotation + dt * effSpin) % 360
    pulse = pulse + dt * effPulse

    local color
    if tint then
        local a = (math.sin(pulse) + 1) / 2
        color = tint:Lerp(Color3.fromRGB(60, 0, 15), 1 - a)
    elseif SystemConfig.RainbowEnabled then
        color = Color3.fromHSV(hue, 1, 1)
    else
        color = STATIC_COLOR
    end

    local alpha = (math.sin(pulse) + 1) / 2
    scale.Scale = 0.85 + alpha * 0.3
    crosshair.Rotation = rotation

    for _, l in ipairs(lines) do l.BackgroundColor3 = color end
    for _, s in ipairs(lineStrokes) do
        s.Enabled = effOutline
        if tintEdge then
            s.Color = tintEdge
            s.Thickness = 2
        else
            s.Color = Color3.fromRGB(0, 0, 0)
            s.Thickness = 1.5
        end
    end
    textStroke.Enabled = SystemConfig.CrosshairTextOutlineEnabled or (tint ~= nil)
    if tint and SystemConfig.CrosshairTextOutlineEnabled then
        textStroke.Color = tintEdge
    end
    crosshairLabel.TextColor3 = color

    local offset = (4 + alpha * 8) + 7
    top.Position    = UDim2.new(0.5,  0,      0.5, -offset)
    bottom.Position = UDim2.new(0.5,  0,      0.5,  offset)
    left.Position   = UDim2.new(0.5, -offset, 0.5,  0)
    right.Position  = UDim2.new(0.5,  offset, 0.5,  0)

    local cam = workspace.CurrentCamera
    local vp = cam and cam.ViewportSize or Vector2.new(1, 1)
    local sx = stickOffset.X / math.max(vp.X, 1)
    local sy = stickOffset.Y / math.max(vp.Y, 1)
    center.Position = UDim2.fromScale(0.5 + sx, 0.5 + sy)
end)
